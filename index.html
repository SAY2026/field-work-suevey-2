📌 How to set up Google Sheets + Apps Script (如何设置Google表格及脚本)

1. Create Google Sheet with these column headers (按以下列名创建表格):
timestamp, fullname, role, roleOther, prevObserve, q4, q5, q6, q7, q8, q9, q10, q11, q12, lyricsAppropriate, studentMusicEngage, compareTraditional, growthPhase, improvementText, scalability, recommendAward, supportItems, strikingMoment, concernsRisks, additionalMsg

2. Open Extensions → Apps Script, paste the code below, then deploy as Web App (Execute as "Me", Access "Anyone").
3. Copy the generated URL and replace GOOGLE_SHEET_API_URL in the HTML above.
4. After deployment, every form submit will add a row to the sheet, and you can also send email feedback using Script's MailApp.

// Apps Script Code (copy into your bound script)
function doPost(e) {
  const sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();
  const headers = ["timestamp","fullname","role","roleOther","prevObserve","q4","q5","q6","q7","q8","q9","q10","q11","q12","lyricsAppropriate","studentMusicEngage","compareTraditional","growthPhase","improvementText","scalability","recommendAward","supportItems","strikingMoment","concernsRisks","additionalMsg"];
  const rowData = headers.map(h => e.parameter[h] || "");
  sheet.appendRow(rowData);
  // OPTIONAL: send feedback to observer if email provided (use MailApp)
  return ContentService.createTextOutput(JSON.stringify({result:"success", row: rowData})).setMimeType(ContentService.MimeType.JSON);
}
function doGet() { return doPost({parameter:{}}); } // simple test
